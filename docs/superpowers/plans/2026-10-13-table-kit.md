**The needle format** (one string per column, the Supplier Portal's encoding):
- text: the typed text, "contains", case-insensitive; `_` reads as a space; `,` and `₱` ignored on both sides.
- enum: `=<value>` — exact (case-insensitive) on the row's value.
- date / amount: `~<op>:<a>[:<b>]`, e.g. `~between:2026-10-01:2026-10-13`, `~ge:1250.50`, `~between::500` (only an upper bound).

```ts
/**
 * The table kit's column description (spec 2026-10-13). Plain data: a server
 * page builds these and hands them to a client table, so nothing here may be a
 * function.
 */

export type ColumnKind = 'text' | 'enum' | 'date' | 'amount'

export const RANGE_OPS = ['eq', 'ne', 'gt', 'ge', 'lt', 'le', 'between'] as const
export type RangeOp = (typeof RANGE_OPS)[number]

export type KitOption = { value: string; label: string }

export type KitColumn = {
  key: string
  label: string
  kind: ColumnKind
  /** enum: the dropdown's values. `value` is what the filter stores; `label` what it shows. */
  options?: readonly KitOption[]
  /** date / amount: the operators the menu offers. Default: all of RANGE_OPS. */
  ops?: readonly RangeOp[]
  /** Cannot be moved to Available Columns. */
  locked?: boolean
  noSort?: boolean
  noFilter?: boolean
  /** Starts in Available Columns. */
  defaultHidden?: boolean
  align?: 'left' | 'right'
  /** Classes for this column's body cells (`KitTable`). */
  className?: string
}

export type KitSort = { key: string; dir: 'asc' | 'desc' } | null

/** Column key → needle. An empty or absent needle is no filter. */
export type KitFilters = Readonly<Record<string, string>>
```

```ts
import { amountToCentavos } from '@/lib/transmittal'
import { isIsoDay } from '@/lib/domain/details'
import { FILTER_MESSAGES, parseAmountBound } from '@/lib/column-filters'
import { RANGE_OPS, type KitColumn, type KitFilters, type KitSort, type RangeOp } from './types'

/**
 * Matching and ordering for the table kit (spec 2026-10-13). Pure and
 * browser-safe — the same imports `lib/column-filters.ts` already makes.
 *
 * Needles, one string per column (the Supplier Portal's encoding):
 *   text    the typed text, "contains"
 *   enum    "=<value>", exact
 *   range   "~<op>:<a>[:<b>]" for date and amount columns
 * A needle that cannot be read REFUSES the whole table (`applyKit` returns
 * nothing), never widens it — the rule the dashboard's filters already follow.
 */

export type RangeNeedle = { op: RangeOp; a: string; b: string }

const RANGE = /^~([a-z]+):([^:]*)(?::([^:]*))?$/

export function parseRangeNeedle(needle: string): RangeNeedle | null {
  const m = RANGE.exec(needle.trim())
  if (!m || !(RANGE_OPS as readonly string[]).includes(m[1])) return null
  return { op: m[1] as RangeOp, a: m[2] ?? '', b: m[3] ?? '' }
}

/** '' when nothing was picked: an empty range is no filter. */
export function rangeNeedle(op: RangeOp, a: string, b = ''): string {
  if (!a && !b) return ''
  return op === 'between' ? `~between:${a}:${b}` : `~${op}:${a}`
}

const isRange = (c: KitColumn) => c.kind === 'date' || c.kind === 'amount'

/** A typed bound as centavos; null when unreadable. Grouping commas allowed. */
const boundCentavos = (v: string): bigint | null => {
  const parsed = parseAmountBound(v)
  return parsed === null ? null : amountToCentavos(parsed)
}

export function needleError(column: KitColumn, needle: string): string | null {
  const n = needle.trim()
  if (!n || !isRange(column)) return null
  const r = parseRangeNeedle(n)
  if (!r) return column.kind === 'date' ? FILTER_MESSAGES.day : FILTER_MESSAGES.amount
  for (const v of [r.a, r.b]) {
    if (!v) continue
    if (column.kind === 'date' && !isIsoDay(v)) return FILTER_MESSAGES.day
    if (column.kind === 'amount' && boundCentavos(v) === null) return FILTER_MESSAGES.amount
  }
  return null
}

const norm = (s: string) => s.toLowerCase().replace(/_/g, ' ').replace(/[,₱]/g, '').replace(/\s+/g, ' ').trim()

function compareBy<T>(op: RangeOp, x: T, a: T | null, b: T | null, lt: (p: T, q: T) => boolean): boolean {
  const eq = (p: T, q: T) => !lt(p, q) && !lt(q, p)
  switch (op) {
    case 'eq': return a === null || eq(x, a)
    case 'ne': return a === null || !eq(x, a)
    case 'gt': return a === null || lt(a, x)
    case 'ge': return a === null || !lt(x, a)
    case 'lt': return a === null || lt(x, a)
    case 'le': return a === null || !lt(a, x)
    case 'between': return (a === null || !lt(x, a)) && (b === null || !lt(b, x))
  }
}

/** Does one row's value satisfy one needle? Assumes `needleError` is null. */
export function matches(column: KitColumn, value: string | null, needle: string): boolean {
  const n = needle.trim()
  if (!n) return true
  if (column.kind === 'date') {
    const r = parseRangeNeedle(n)
    const d = value?.trim().slice(0, 10) ?? ''
    if (!r || !d) return false
    return compareBy(r.op, d, r.a || null, r.b || null, (p, q) => p < q)
  }
  if (column.kind === 'amount') {
    const r = parseRangeNeedle(n)
    const x = value === null ? null : amountToCentavos(value.trim())
    if (!r || x === null) return false
    const a = r.a ? boundCentavos(r.a) : null
    const b = r.b ? boundCentavos(r.b) : null
    return compareBy(r.op, x, a, b, (p, q) => p < q)
  }
  if (value === null) return false
  if (n.startsWith('=')) return norm(value) === norm(n.slice(1))
  return norm(value).includes(norm(n))
}

function compareValues(column: KitColumn, a: string, b: string): number {
  if (column.kind === 'amount') {
    const x = amountToCentavos(a)
    const y = amountToCentavos(b)
    if (x !== null && y !== null) return x < y ? -1 : x > y ? 1 : 0
  }
  return a.localeCompare(b, 'en', { numeric: true, sensitivity: 'base' })
}

/**
 * The rows a table shows: every active filter, then the sort. Empty values
 * sort last in both directions; ties keep their original order.
 */
export function applyKit<R>(
  rows: readonly R[],
  columns: readonly KitColumn[],
  valueOf: (row: R, key: string) => string | null,
  filters: KitFilters,
  sort: KitSort,
): R[] {
  const byKey = new Map(columns.map((c) => [c.key, c]))
  const active = Object.entries(filters).filter(([k, v]) => v.trim() !== '' && byKey.has(k))
  if (active.some(([k, v]) => needleError(byKey.get(k)!, v) !== null)) return []
  const kept = rows.filter((r) => active.every(([k, v]) => matches(byKey.get(k)!, valueOf(r, k), v)))
  const col = sort ? byKey.get(sort.key) : undefined
  if (!sort || !col) return kept
  const sign = sort.dir === 'desc' ? -1 : 1
  return kept
    .map((r, i) => [r, i] as const)
    .sort(([x, xi], [y, yi]) => {
      const a = valueOf(x, sort.key)?.trim() ?? ''
      const b = valueOf(y, sort.key)?.trim() ?? ''
      if (!a || !b) return !a && !b ? xi - yi : !a ? 1 : -1
      return sign * compareValues(col, a, b) || xi - yi
    })
    .map(([r]) => r)
}

const MANILA_MS = 8 * 60 * 60 * 1000

/** The Manila calendar day of an instant. */
export function manilaDay(d: Date): string {
  return new Date(d.getTime() + MANILA_MS).toISOString().slice(0, 10)
}

export function manilaToday(now: Date = new Date()): string {
  return manilaDay(now)
}

export function thisMonth(day: string): [string, string] {
  const [y, m] = day.split('-').map(Number)
  const last = new Date(Date.UTC(y, m, 0)).getUTCDate()
  return [`${day.slice(0, 8)}01`, `${day.slice(0, 8)}${String(last).padStart(2, '0')}`]
}

export function shiftDay(day: string, by: number): string {
  const d = new Date(`${day}T00:00:00Z`)
  d.setUTCDate(d.getUTCDate() + by)
  return d.toISOString().slice(0, 10)
}

/** A stored calendar date (`@db.Date`, UTC midnight) as the row value. */
export const dateValue = (d: Date | null): string | null => (d ? d.toISOString().slice(0, 10) : null)

/** An instant (a timestamp) as the row value: its Manila day. */
export const instantValue = (d: Date | null): string | null => (d ? manilaDay(d) : null)

const OP_WORD: Record<Exclude<RangeOp, 'between'>, string> = { eq: '=', ne: '≠', gt: '>', ge: '≥', lt: '<', le: '≤' }

/** A chip's text for a needle. */
export function humanNeedle(column: KitColumn, needle: string): string {
  const n = needle.trim()
  const r = isRange(column) ? parseRangeNeedle(n) : null
  if (r) {
    if (r.op !== 'between') return `${OP_WORD[r.op]} ${r.a}`
    if (r.a && r.b) return r.a === r.b ? r.a : `${r.a} TO ${r.b}`
    return r.a ? `FROM ${r.a}` : `UNTIL ${r.b}`
  }
  if (n.startsWith('=')) {
    const v = n.slice(1)
    return column.options?.find((o) => o.value === v)?.label ?? v
  }
  return n
}
```

```ts
import type { KitColumn } from './types'

/**
 * A table's presentation, per browser (spec 2026-10-13 §1): the order of every
 * column, which are hidden, and any dragged widths. Pure; the hook in
 * `components/table-kit/useTableLayout.ts` does the storage I/O.
 *
 * Whatever storage holds is read defensively: a stored layout outlives
 * releases, so unknown keys are dropped, columns added since are inserted
 * beside their usual neighbour, a locked column is never hidden, and at least
 * one column always shows.
 */

export type Layout = { order: string[]; hidden: string[]; widths: Record<string, number> }

export const MIN_WIDTH = 60
export const MAX_WIDTH = 800

export const clampWidth = (w: number) => Math.min(MAX_WIDTH, Math.max(MIN_WIDTH, Math.round(w)))

export const layoutStorageKey = (tableId: string) => `check-monitoring.table.${tableId}.v1`

export function defaultLayout(columns: readonly KitColumn[]): Layout {
  return {
    order: columns.map((c) => c.key),
    hidden: columns.filter((c) => c.defaultHidden && !c.locked).map((c) => c.key),
    widths: {},
  }
}

export function sanitizeLayout(raw: unknown, columns: readonly KitColumn[]): Layout {
  const def = defaultLayout(columns)
  if (raw === null || typeof raw !== 'object' || Array.isArray(raw)) return def
  const r = raw as Record<string, unknown>
  if (!Array.isArray(r.order)) return def
  const keys = columns.map((c) => c.key)

  const stored: string[] = []
  for (const k of r.order) if (typeof k === 'string' && keys.includes(k) && !stored.includes(k)) stored.push(k)
  if (stored.length === 0) return def

  const order = [...stored]
  const added: string[] = []
  for (const k of keys) {
    if (order.includes(k)) continue
    const before = keys.slice(0, keys.indexOf(k)).reverse().find((p) => order.includes(p))
    order.splice(before === undefined ? 0 : order.indexOf(before) + 1, 0, k)
    added.push(k)
  }

  const locked = new Set(columns.filter((c) => c.locked).map((c) => c.key))
  const hidden: string[] = []
  for (const k of Array.isArray(r.hidden) ? r.hidden : []) {
    if (typeof k === 'string' && stored.includes(k) && !locked.has(k) && !hidden.includes(k)) hidden.push(k)
  }
  for (const k of added) if (def.hidden.includes(k)) hidden.push(k)

  const widths: Record<string, number> = {}
  if (r.widths !== null && typeof r.widths === 'object') {
    for (const [k, v] of Object.entries(r.widths as Record<string, unknown>)) {
      if (keys.includes(k) && typeof v === 'number' && Number.isFinite(v)) widths[k] = clampWidth(v)
    }
  }
  return { order, hidden: hidden.length >= keys.length ? [] : hidden, widths }
}

function safeParse(raw: string | null): unknown {
  if (!raw) return null
  try { return JSON.parse(raw) } catch { return null }
}

export function loadLayout(
  storage: Pick<Storage, 'getItem'>,
  tableId: string,
  columns: readonly KitColumn[],
  legacy?: (s: Pick<Storage, 'getItem'>) => unknown,
): Layout {
  const own = safeParse(storage.getItem(layoutStorageKey(tableId)))
  return sanitizeLayout(own ?? (legacy ? legacy(storage) : null), columns)
}

/** A stored JSON array of the VISIBLE keys in order (the main list's `columns.v3`) as a layout. */
export function fromVisibleList(raw: string | null, columns: readonly KitColumn[]): Layout | null {
  const parsed = safeParse(raw)
  if (!Array.isArray(parsed)) return null
  const keys = columns.map((c) => c.key)
  const visible = [...new Set(parsed.filter((k): k is string => typeof k === 'string' && keys.includes(k)))]
  if (visible.length === 0) return null
  const rest = keys.filter((k) => !visible.includes(k))
  return { order: [...visible, ...rest], hidden: rest, widths: {} }
}

/** The columns on screen, in order; a forced (filtered) column shows even when hidden. */
export function visibleColumns(columns: readonly KitColumn[], layout: Layout, forced: readonly string[] = []): KitColumn[] {
  const byKey = new Map(columns.map((c) => [c.key, c]))
  return layout.order
    .filter((k) => !layout.hidden.includes(k) || forced.includes(k))
    .map((k) => byKey.get(k))
    .filter((c): c is KitColumn => c !== undefined)
}

/** Shown again: it joins the end of the selected list. */
export function showColumn(layout: Layout, key: string): Layout {
  if (!layout.hidden.includes(key)) return layout
  return { ...layout, order: [...layout.order.filter((k) => k !== key), key], hidden: layout.hidden.filter((k) => k !== key) }
}

export function hideColumn(layout: Layout, key: string, columns: readonly KitColumn[]): Layout {
  const visible = layout.order.filter((k) => !layout.hidden.includes(k))
  const locked = columns.some((c) => c.key === key && c.locked)
  if (locked || !visible.includes(key) || visible.length <= 1) return layout
  return { ...layout, hidden: [...layout.hidden, key] }
}

/** Swap with the visible neighbour; hidden columns keep their slots. */
export function moveVisible(layout: Layout, key: string, by: -1 | 1): Layout {
  const visible = layout.order.filter((k) => !layout.hidden.includes(k))
  const i = visible.indexOf(key)
  const j = i + by
  if (i < 0 || j < 0 || j >= visible.length) return layout
  const order = [...layout.order]
  const a = order.indexOf(visible[i])
  const b = order.indexOf(visible[j])
  ;[order[a], order[b]] = [order[b], order[a]]
  return { ...layout, order }
}

/** A dragged width, or `null` to forget it (double-click on the grip). */
export function setWidth(layout: Layout, key: string, width: number | null): Layout {
  const widths = { ...layout.widths }
  if (width === null) delete widths[key]
  else widths[key] = clampWidth(width)
  return { ...layout, widths }
}

/** Inline style for a sized column's header and body cells. */
export const widthStyle = (w?: number) => (w ? { width: w, minWidth: w, maxWidth: w } : undefined)
```

```ts
import { parseRangeNeedle, rangeNeedle, shiftDay } from './match'
import type { RangeOp } from './types'

/**
 * How a column's needle is written into the URL parameters a server page
 * already parses (spec 2026-10-13 §3, §5). Nothing new is invented on the
 * server: a day range is the page's inclusive From/To, an amount range its
 * Min/Max, so the operators offered are only those that map onto them.
 * Typed text passes through untouched — the server's parser decides whether it
 * can be read, and refuses if not.
 */

export type Binding =
  | { kind: 'text'; param: string }
  | { kind: 'enum'; param: string }
  | { kind: 'range'; from: string; to: string; unit: 'day' | 'amount' }

/** "Does Not Equal" has no From/To; strict bounds on amounts have no Min/Max. */
export const SERVER_DAY_OPS: readonly RangeOp[] = ['eq', 'gt', 'ge', 'lt', 'le', 'between']
export const SERVER_AMOUNT_OPS: readonly RangeOp[] = ['eq', 'ge', 'le', 'between']

export function bindingParams(b: Binding): string[] {
  return b.kind === 'range' ? [b.from, b.to] : [b.param]
}

export function needleToParams(b: Binding, needle: string): Record<string, string> {
  const n = needle.trim()
  if (b.kind === 'text') return { [b.param]: n }
  if (b.kind === 'enum') return { [b.param]: n.replace(/^=/, '') }
  const r = parseRangeNeedle(n)
  let from = ''
  let to = ''
  if (r) {
    const next = (d: string, by: number) => (b.unit === 'day' && d ? shiftDay(d, by) : d)
    switch (r.op) {
      case 'eq': from = r.a; to = r.a; break
      case 'ge': from = r.a; break
      case 'le': to = r.a; break
      case 'gt': from = next(r.a, 1); break
      case 'lt': to = next(r.a, -1); break
      case 'between': from = r.a; to = r.b; break
      case 'ne': break
    }
  }
  return { [b.from]: from, [b.to]: to }
}

export function paramsToNeedle(b: Binding, values: Readonly<Record<string, string>>): string {
  if (b.kind === 'text') return values[b.param]?.trim() ?? ''
  if (b.kind === 'enum') {
    const v = values[b.param]?.trim() ?? ''
    return v ? `=${v}` : ''
  }
  const from = values[b.from]?.trim() ?? ''
  const to = values[b.to]?.trim() ?? ''
  if (from && to) return from === to ? rangeNeedle('eq', from) : rangeNeedle('between', from, to)
  if (from) return rangeNeedle('ge', from)
  if (to) return rangeNeedle('le', to)
  return ''
}

/** The server's refusal of any of this binding's parameters. */
export function bindingError(b: Binding, errors: Readonly<Record<string, string>>): string | undefined {
  return bindingParams(b).map((p) => errors[p]).find(Boolean)
}
```

```ts
'use client'

import { useEffect, useLayoutEffect, useState } from 'react'
import type { KitColumn } from '@/lib/table-kit/types'
import { defaultLayout, layoutStorageKey, loadLayout, type Layout } from '@/lib/table-kit/layout'

/** Runs before paint on the client; an ordinary effect on the server, where layout effects warn. */
const useBeforePaint = typeof window === 'undefined' ? useEffect : useLayoutEffect

/**
 * One table's layout, per browser. Starts at the default so the server render
 * and the first client render agree, and is corrected before paint. Storage
 * that throws (a private window, blocked site data) leaves the default.
 */
export function useTableLayout(
  tableId: string,
  columns: readonly KitColumn[],
  legacy?: (s: Pick<Storage, 'getItem'>) => unknown,
) {
  const [layout, setLayout] = useState<Layout>(() => defaultLayout(columns))

  useBeforePaint(() => {
    try {
      setLayout(loadLayout(window.localStorage, tableId, columns, legacy))
    } catch {
      // Blocked storage: the default stays.
    }
    // Columns are rebuilt every render; the layout is read once per table.
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [tableId])

  const save = (next: Layout) => {
    setLayout(next)
    try { window.localStorage.setItem(layoutStorageKey(tableId), JSON.stringify(next)) } catch { /* not remembered */ }
  }
  const reset = () => {
    setLayout(defaultLayout(columns))
    try { window.localStorage.removeItem(layoutStorageKey(tableId)) } catch { /* nothing stored */ }
  }
  return { layout, save, reset }
}
```

```ts
'use client'

import { usePathname, useRouter, useSearchParams } from 'next/navigation'
import { filterHref } from '@/lib/filter-href'
import { markOwnNavigation } from '../filter-nav'

/**
 * Rewrite some of the URL's parameters and navigate softly (spec 2026-10-13 §2).
 * `updates` maps a parameter to its new value; '' removes it. `drop` names
 * parameters that any change invalidates (Audit's `before` cursor).
 *
 * `markOwn`: true when nothing else on the page shows these parameters in an
 * uncontrolled control (the main list — its hidden inputs are controlled), so
 * the filter bar is not remounted. False where the page's own filter bar shows
 * the same parameter (Audit's ACTION and USER selects): the bar must remount
 * to show the menu's choice.
 */
export function useUrlUpdate(opts: { markOwn: boolean; drop?: readonly string[] }) {
  const router = useRouter()
  const pathname = usePathname()
  const search = useSearchParams()
  return (updates: Readonly<Record<string, string>>) => {
    const entries = [...search.entries()].filter(([k]) => !(k in updates) && !(opts.drop ?? []).includes(k))
    for (const [k, v] of Object.entries(updates)) if (v) entries.push([k, v])
    const href = filterHref(entries, pathname)
    if (opts.markOwn) markOwnNavigation(href.includes('?') ? href.slice(href.indexOf('?') + 1) : '')
    router.replace(href, { scroll: false })
  }
}
```

```tsx
'use client'

import { useRef } from 'react'
import { clampWidth } from '@/lib/table-kit/layout'

/**
 * Drag a header's right edge to size its column; double-click forgets the
 * width. Pointer capture keeps the drag alive once the pointer leaves the grip.
 */
export function ResizeGrip({ label, width, onResize }: { label: string; width?: number; onResize: (width: number | null) => void }) {
  const drag = useRef<{ x: number; w: number } | null>(null)
  return (
    <div
      role="separator" aria-orientation="vertical" aria-label={`Resize ${label}`}
      onPointerDown={(e) => {
        const th = e.currentTarget.parentElement
        drag.current = { x: e.clientX, w: width ?? th?.getBoundingClientRect().width ?? 120 }
        e.currentTarget.setPointerCapture(e.pointerId)
        e.preventDefault()
        e.stopPropagation()
      }}
      onPointerMove={(e) => {
        const d = drag.current
        if (d) onResize(clampWidth(d.w + e.clientX - d.x))
      }}
      onPointerUp={() => { drag.current = null }}
      onPointerCancel={() => { drag.current = null }}
      onDoubleClick={(e) => { e.stopPropagation(); onResize(null) }}
      onClick={(e) => e.stopPropagation()}
      className="absolute right-0 top-0 h-full w-2 cursor-col-resize touch-none hover:bg-navy/20"
    />
  )
}
```

```tsx
'use client'

import type { KitColumn, KitFilters } from '@/lib/table-kit/types'
import { humanNeedle } from '@/lib/table-kit/match'

const n = (x: number) => x.toLocaleString('en-PH')

/** One chip per active filter, each with ✕; the row count; CLEAR FILTERS. Nothing when no filter is set. */
export function FilterChips({
  columns, filters, errors = {}, onClear, onClearAll, count, note,
}: {
  columns: readonly KitColumn[]
  filters: KitFilters
  errors?: Readonly<Record<string, string>>
  onClear: (key: string) => void
  onClearAll?: () => void
  /** Client mode: rows shown of rows held; `capped` when the page drew only some of the set. */
  count?: { shown: number; total: number; capped: boolean }
  /** A line of its own, e.g. why NUMBERING's MISSING lines are not on screen. */
  note?: string
}) {
  const active = columns.filter((c) => (filters[c.key] ?? '').trim() !== '')
```

```ts
import type { KitColumn } from '@/lib/table-kit/types'

/**
 * The transmittal's pick list (client, 2026-10-09: "include also filter and
 * customizable and adjustable columns"). Since 2026-10-13 the filtering,
 * sorting and column layout are the table kit's (`lib/table-kit/`), the same

 * Amounts stay decimal strings and compare in centavos (rule 8).
 */

export type TransmittalStatus = 'SIGNATURE_PENDING' | 'SIGNED' | 'RELEASED'

export type TransmittalCandidate = {
  id: string
  checkNumber: string
  cashAccount: string
  poNumber: string
  voucher: string
  payee: string
  amount: string | null
  currency: string
  status: TransmittalStatus
  company: string
}

export const PICK_COLUMNS = [
  'checkNumber', 'cashAccount', 'poNumber', 'voucher', 'payee', 'company', 'status', 'amount',
] as const
export type PickColumn = (typeof PICK_COLUMNS)[number]

export const PICK_LABELS: Record<PickColumn, string> = {
  checkNumber: 'CHECK NUMBER',
  cashAccount: 'CASH ACCOUNT',
  poNumber: 'PO NUMBER / VENDOR REF',
  voucher: 'VOUCHER NUMBER',
  payee: 'PAYEE',
  company: 'COMPANY',
  status: 'STATUS',
  amount: 'AMOUNT',
}

export const TRANSMITTAL_TABLE_ID = 'transmittal'
/** The pre-kit preference; its `{ order, hidden, widths }` shape is the kit's layout, read once. */
export const LEGACY_PREFS_KEY = 'check-monitoring.transmittal.columns.v1'

/** The sheet's own order first, then the extras the sheet does not carry. */
export function pickColumns(companies: readonly string[]): KitColumn[] {
  const text = (key: PickColumn): KitColumn => ({ key, label: PICK_LABELS[key], kind: 'text' })
  return [
    { ...text('checkNumber'), className: 'font-medium' },
    text('cashAccount'),
    text('poNumber'),
    text('voucher'),
    text('payee'),
    { key: 'amount', label: PICK_LABELS.amount, kind: 'amount', align: 'right', className: 'tabular-nums' },
    { key: 'company', label: PICK_LABELS.company, kind: 'enum', options: companies.map((c) => ({ value: c, label: c })) },
    // The status choice is the dropdown above the table.
    { key: 'status', label: PICK_LABELS.status, kind: 'enum', noFilter: true },
  ]
}

export function pickValue(c: TransmittalCandidate, key: string): string | null {
  switch (key as PickColumn) {
    case 'amount': return c.amount
    case 'status': return c.status.replace(/_/g, ' ')
    default: return c[key as Exclude<PickColumn, 'amount' | 'status'>] || null
  }
}

export type StatusChoice = 'ALL' | TransmittalStatus

/** ALL is SIGNATURE PENDING + SIGNED — RELEASED only on its own choice. */
export function byStatus(rows: readonly TransmittalCandidate[], status: StatusChoice): TransmittalCandidate[] {
  return rows.filter((c) => (status === 'ALL' ? c.status !== 'RELEASED' : c.status === status))
}
```

```tsx
'use client'

import type { ReactNode } from 'react'
import { DEFAULT_COLUMNS, withColumnOrder } from '@/lib/table-columns'
import { readMainListOrder } from '@/lib/table-kit/main-list'

/**
 * EXPORT EXCEL, in the viewer's column order (spec C3; since 2026-10-13 the
 * table kit's layout). The order lives only in this browser's `localStorage`,
 * so it is added to the link at click time. Without JavaScript — or with
 * storage blocked — the plain link exports in the default order. A modified
 * click (new tab, save link) is left alone.
 */
export function ExportLink({ href, className, children }: { href: string; className: string; children: ReactNode }) {
  return (
    <a
      href={href}
      className={className}
      onClick={(e) => {
        if (e.button !== 0 || e.metaKey || e.ctrlKey || e.shiftKey || e.altKey) return
        let order = DEFAULT_COLUMNS
        try {
          order = readMainListOrder(window.localStorage)
        } catch {
          // Blocked storage: the default order.
        }
        e.preventDefault()
        window.location.assign(withColumnOrder(href, order))
      }}
    >
      {children}
    </a>
  )
}
```

```tsx
  tableId, columns, rows, bindings, values, drop, markOwn, empty = 'NO ROWS MATCH.',
}: {
  tableId: string
  columns: readonly KitColumn[]
  rows: readonly KitRow[]
  bindings: Readonly<Record<string, Binding>>
  /** The page's parameters as it read them. */
  values: Readonly<Record<string, string>>
  /** Parameters any filter change invalidates (a pagination cursor). */
  drop?: readonly string[]
  markOwn: boolean
  empty?: string
}) {
  const { layout, save, reset } = useTableLayout(tableId, columns)
  const update = useUrlUpdate({ markOwn, drop })
  const needles: Record<string, string> = {}
  for (const c of columns) if (bindings[c.key]) needles[c.key] = paramsToNeedle(bindings[c.key], values)
  const filtered = Object.keys(needles).filter((k) => needles[k] !== '')
  const visible = visibleColumns(columns, layout, filtered)
  const apply = (key: string, needle: string) => { if (bindings[key]) update(needleToParams(bindings[key], needle)) }

  return (
    <div className="space-y-2">
      <div className="flex flex-wrap items-center gap-2">
        <ColumnConfigButton columns={columns} layout={layout} onSave={save} onReset={reset} />
        <FilterChips
          columns={columns} filters={needles} onClear={(k) => apply(k, '')}
          onClearAll={() => update(Object.fromEntries(Object.values(bindings).flatMap(bindingParams).map((p) => [p, ''])))}
        />
      </div>
      <div className="overflow-x-auto rounded-2xl bg-white ring-1 ring-hairline">
        <table className="w-full text-sm">
          <thead className="border-b border-hairline text-left text-[11px] font-semibold tracking-widest text-slate-400">
            <KitHeaderRow
              columns={visible} widths={layout.widths} sort={null} filters={needles}
              onSort={() => {}} onFilter={apply} onResize={(key, w) => save(setWidth(layout, key, w))}
            />
          </thead>
          <tbody>
            {rows.length === 0 && <tr><td colSpan={visible.length} className="p-8 text-center text-sm text-slate-500">{empty}</td></tr>}
            {rows.map((r) => 'band' in r ? null : (
              <tr key={r.key} className={r.className ?? 'border-b border-slate-100 last:border-0 odd:bg-white even:bg-ground'}>
                {visible.map((c) => (

                ))}
              </tr>
            ))}
          </tbody>
        </table>
      </div>
```

### Conversion recipe for Tasks 8–11 (client mode)

Every remaining table becomes a `KitTable`. The page (server or client) keeps rendering its cells;
it only reshapes them:

1. Declare `columns: KitColumn[]` — one per `<th>`, same order, same label. `kind` by content: dates `date`, money and counts `amount`, a closed set of words (status, company, tenant, mode) `enum` with `options` built from the distinct values present (`[...new Set(rows.map(…))].sort().map((v) => ({ value: v, label: v }))`), anything else `text`. Right-aligned `<th>`s get `align: 'right'`. A `<td>` with classes beyond the padding moves those classes to `className`. A column of buttons or forms gets `noSort: true, noFilter: true, locked: true`.
2. Map each row to `{ key, values, cells }`: `cells[key]` is exactly the old `<td>`'s CHILDREN; `values[key]` is the plain string the reader sees — `dateValue(d)` for a stored date, `instantValue(d)` for a timestamp, the decimal string for money (`amount.toString()` on a Prisma `Decimal`, or the string the row already holds), `String(n)` for a count, `null` where the cell shows `—`.
3. Render `<KitTable tableId="<id>" columns={columns} rows={rows} … />` in place of the `<section>/<div>` + `<table>`. Pass `capped` when the page drew fewer rows than exist, `footer` for a totals row, `frameClassName` when the old wrapper differed from the default.
4. Import from `@/components/table-kit/KitTable` (`KitTable`, `type KitRow`), `@/lib/table-kit/types` (`type KitColumn`), `@/lib/table-kit/match` (`dateValue`, `instantValue`).
5. After each task: `node node_modules/typescript/bin/tsc --noEmit`, the named view tests, a browser look at each page (sort one column, filter one, hide one, resize one, reload), `read_console_messages` errors none, commit.
